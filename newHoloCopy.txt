float GetSPDCorrection(float wavelength)
{
    // Use a high-accuracy spectral power distribution correction function
    // Based on research-quality data and scientific models
    return 1.0 + 0.1 * sin(wavelength * 2 * 3.14159 / 0.6328);
}



float3x3 CreateSkewMatrix(float3 tangent, float amount)
{
    float3 skewAxis = normalize(cross(tangent, float3(0, 1, 0))); // Assumes Y is up
    float3x3 skewMatrix = float3x3(1, 0, 0,
                                   0, 1, 0,
                                   0, 0, 1);

    skewMatrix[0][2] = skewAxis.x * amount; // Skew along the X-axis of the tangent space 
    skewMatrix[1][2] = skewAxis.y * amount; // Skew along the Y-axis of the tangent space
    skewMatrix[2][2] = skewAxis.z * amount; // Skew along the Z-axis of the tangent space

    return skewMatrix;
}

float3 skewNormalX(float3 normal, float amount)
{
    return normalize(mul(normal, CreateSkewMatrix(normal, amount))); // Added normalization
}

// Improved gradient function with a larger set of gradient vectors
inline float gradient(uint hash, float3 p)
{
    static const float3 gradients[16] =
    {
        float3(1, 1, 0), float3(-1, 1, 0), float3(1, -1, 0), float3(-1, -1, 0),
        float3(1, 0, 1), float3(-1, 0, 1), float3(1, 0, -1), float3(-1, 0, -1),
        float3(0, 1, 1), float3(0, -1, 1), float3(0, 1, -1), float3(0, -1, -1),
        float3(1, 1, 1), float3(-1, 1, 1), float3(1, -1, 1), float3(-1, -1, 1)
    };

    return dot(gradients[hash & 16], p);
}


float gradient(uint hash, float3 f, float sx, float cy, float tz)
{
    // Construct a pseudo-random gradient vector based on the hash value
    float u = hash * 0.1031;
    float v = hash * 0.11113;
    float w = hash * 0.13287;
    
    // Apply the gradient vector to the fractional coordinates
    return (u * f.x + v * f.y + w * f.z) * (sx + cy + tz);
}


inline float slerp(float q1, float q2, float t)
{
    // Ensure the shortest arc for interpolation
    float cosOmega = dot(q1, q2);
    if (cosOmega < 0.0)
    {
        q2 = -q2;
        cosOmega = -cosOmega;
    }

    // Calculate interpolation factors based on angle
    float k0, k1;
    if (cosOmega > 0.9999)
    {
        // Small angles: approximate linear interpolation
        k0 = 1.0 - t;
        k1 = t;
    }
    else
    {
        // Standard spherical linear interpolation
        float sinOmega = sqrt(1.0 - cosOmega * cosOmega);
        float omega = atan2(sinOmega, cosOmega);
        float inv_sin_omega = 1.0 / sinOmega;
        k0 = sin((1.0 - t) * omega) * inv_sin_omega;
        k1 = sin(t * omega) * inv_sin_omega;
    }

    return k0 * q1 + k1 * q2; // Interpolated result
}

inline float lerp(float x, float x0, float x1, float y0, float y1)
{
    if (abs(x1 - x0) < epsilon) // Added check for near-zero difference
        return y0;
    return y0 + (x - x0) * (y1 - y0) / (x1 - x0);
}

// Procedural Noise
inline float pnoise(float3 uvw, float time)
{
    float2 noiseScale = float2(1.0, 1.0);
    float2 noiseOffset = float2(0.0, 0.0);
    float frequency = 1.0;
    float amplitude = 1.0;
    float octaves = 6.0;
  
    float2 uvNoise = uvw.xy * noiseScale + noiseOffset;
    float noisev = 0.0;
  
    for (int i = 0; i < octaves; i++)
    {
        noisev += sin((uvNoise.x + time * frequency + uvw.z * 10.0) * (1 << i)) * amplitude;
        amplitude *= 0.5;
    }
  
    return noisev;
}

float3 interpolate(float3 a, float3 b, float3 t)
{
    float3 result;
    result.x = a.x + t.x * (b.x - a.x);
    result.y = a.y + t.y * (b.y - a.y);
    result.z = a.z + t.z * (b.z - a.z);

    // Apply a custom interpolation curve
    result.x = 3.0 * result.x * result.x - 2.0 * result.x * result.x * result.x;
    result.y = 3.0 * result.y * result.y - 2.0 * result.y * result.y * result.y;
    result.z = 3.0 * result.z * result.z - 2.0 * result.z * result.z * result.z;
    
    return result;
}

float4 interpolate(float4 a, float4 b, float4 t)
{
    float4 result;
    result.x = a.x + t.x * (b.x - a.x);
    result.y = a.y + t.y * (b.y - a.y);
    result.z = a.z + t.z * (b.z - a.z);
    result.w = a.w + t.w * (b.w - a.w);
    // Apply a custom interpolation curve
    result.x = 3.0 * result.x * result.x - 2.0 * result.x * result.x * result.x;
    result.y = 3.0 * result.y * result.y - 2.0 * result.y * result.y * result.y;
    result.z = 3.0 * result.z * result.z - 2.0 * result.z * result.z * result.z;
    result.w = 3.0 * result.w * result.w - 2.0 * result.w * result.w * result.w;
    
    return result;
}


float Gaussian(float x, float sigma)
{
    return exp(-0.5 * (x * x) / (sigma * sigma)) / (sqrt(2.0 * 3.14159265358979323846) * sigma);
}

float Gaussian(float x, float y, float sigma)
{
    return exp(-0.5 * (x * y) / (sigma * sigma)) / (sigma * sqrt(2.0 * 3.14159265358979323846));
}

float4 GaussianBilinear(float4 p00, float4 p11, float2 uv)
{
    float2 uv2 = uv * uv;
    float4 a = slerp(p00, p11, uv2.x);
    float4 b = slerp(p00, p11, 1.0 - uv2.x);
    return slerp(a, b, saturate(sqrt(uv2.x + uv2.y)));
}

float4 RadialBilinear(float4 p00, float4 p11, float2 uv)
{
    float2 uvLength = length(uv * 2.0 - 1.0);
    float4 a = slerp(p00, p11, uvLength.x);
    float4 b = slerp(p00, p11, uvLength.y);
    return slerp(a, b, uvLength.x * uvLength.y);
}

float4 SinusoidalBilinear(float4 p00, float4 p11, float2 uv)
{
    float2 uvSin = sin(uv * 3.14159 * 2.0);
    float4 a = slerp(p00, p11, uvSin.x);
    float4 b = slerp(p00, p11, uvSin.y);
    return slerp(a, b, uvSin.x * uvSin.y);
}

float4 FractalBilinear(float4 p00, float4 p11, float2 uv)
{
    float2 uvFractal = uv;
    float4 result = p00;
    float weight = 1.0;
    for (int i = 0; i < 4; i++)
    {
        float4 a = slerp(p00, p11, uvFractal.x);
        float4 b = slerp(p00, p11, uvFractal.y);
        result += slerp(a, b, uvFractal.x * uvFractal.y) * weight;
        uvFractal *= 2.0;
        weight *= 0.5;
    }
    return result;
}

float4 HolographicBilearWithCaustics(float4 p00, float4 p11, float2 uv, float time)
{
    float2 holographicUV = uv + float2(
config.depth * sin(uv.x * 10.0 + time * 2.0) * 0.01,
config.depth * cos(uv.y * 20.0 + time * 3.0) * 0.01
    );
    
    float4 a = config.depth * GaussianBilinear(p00, p11, holographicUV);
    float4 b = config.depth * GaussianBilinear(p00, p11, holographicUV + float2(sin(time * 5.0), cos(time * 7.0)) * 0.02);
    float4 c = config.depth * GaussianBilinear(p00, p11, holographicUV + float2(sin(time * 10.0), cos(time * 13.0)) * 0.03);
    
    float causticIntensity = config.depth * .01 * sin(uv.x * 50.0 + time * 10.0) * sin(uv.y * 70.0 + time * 15.0) * 0.5 + 0.5;
    return (a + b + c) / 3.0 * causticIntensity * .05;
}

float4 RefractedHolographicBilinear(float4 p00, float4 p11, float2 uv, float time)
{
    float2 holographicUV = uv + float2(
        sin(uv.x * 2.0 + time * 2.0) * 0.01,
        cos(uv.y * 3.0 + time * 3.0) * 0.01
    ) * pnoise(vec3(uv, 0), time);
    
    float2 refractedUV = holographicUV + float2(
        sin(holographicUV.x * 5.0 + time * 3.0) * 0.02,
        cos(holographicUV.y * 7.0 + time * 5.0) * 0.02
    ) * pnoise(vec3(uv, 1), time);
    
    float4 a = GaussianBilinear(p00, p11, refractedUV);
    float4 b = GaussianBilinear(p00, p11, refractedUV + float2(sin(time * 2.0), cos(time * 3.0)) * 0.03);
    float4 c = GaussianBilinear(p00, p11, refractedUV + float2(sin(time * 3.0), cos(time * 5.0)) * 0.04);
    
    return (a + b + c) / 3.0;
}

float4 VolumetricHolographicBilinear(float4 p00, float4 p11, float2 uv, float time)
{
    // Add some noise to the uv coordinates to give it a more organic feel
    float2 noisyUV = uv + float2(
        sin(uv.x * 3.0 + time * 2.0) / 2 * 0.005 * config.depth * (PerlinNoise(uv, time)),
        cos(uv.y * 5.0 + time * 3.0) / 2 * 0.003 * config.depth * (PerlinNoise(uv, time))
    );
    
    // Use a more complex refracting function to create a sense of depth
    float2 refractedUV = noisyUV + float2(
        sin(noisyUV.x + time) * 0.004,
        cos(noisyUV.y + time) * 0.002
    );
    
    // Use a combination of spherical and cylindrical coordinates to create a more intricate pattern
    float4 a = slerp(p00, p11, refractedUV.x);
    float4 b = slerp(p00, p11, refractedUV.x + sin(time * 3.0) * 0.005);
    float4 c = slerp(p00, p11, refractedUV.y);
    float4 d = slerp(p00, p11, refractedUV.y + sin(time * 7.0) * 0.009);
    
    // Combine the results using a more complex blending function
    return slerp(slerp(p00, RefractedHolographicBilinear(a, b, uv / 2, time) * (config.depth * .15 + .75), .5),
            slerp(p11, RefractedHolographicBilinear(c, d, uv * 2, time) * (config.depth * .15 + .5), .5),
    .5)
    + .01 * float4(
        sin(uv.x * 3.0 + time) * 0.005 * PerlinNoise(uv, time / 3),
        cos(uv.y * 5.0 + time) * 0.005 * PerlinNoise(uv, time / 5),
        sin(uv.x * 7.0 + time) * 0.005 * PerlinNoise(uv, time),
        cos(uv.y * 9.0 + time) * 0.005 * PerlinNoise(uv, time)
    );
}

// Example usage
float3 NoiseGenerator(float3 position, float scale, float frequency, float amplitude, float lacunarity, float persistence)
{
    float3 fnoise = float3(0.0, 0.0, 0.0);
    float totalAmplitude = 0.0;
    
    for (int i = 0; i < 6; i++)
    {
        fnoise += PerlinNoise(position * frequency) * amplitude;
        totalAmplitude += amplitude;
        frequency *= lacunarity;
        amplitude *= persistence;
    }
    
    return fnoise / totalAmplitude;
}

// Helper function to create a GDMFConfig struct

{
    GDMFConfig config;
    config.nearPoint = nearPoint;

    config.transitionPoint = transitionPoint;
    config.polynomialDegree = polynomialDegree;

    config.skewFactor = skewFactor;

    config.nearSlope = nearSlope;
    config.farSlope = farSlope;
    config.transitionWidth = transitionWidth;
    return config;
}

float EvaluatePolynomial(float x, float degree, float weight)
{
    float result = 0.0f;
    for (int i = 0; i <= degree; ++i)
    {
        result += weight * pow(x, i) / pow(2.0f, i);
    }
    return result;
}
// Generalized Depth Mapping Function
float GDMF(float depth, GDMFConfig config)
{
    float normalizedDepth = (depth - config.nearPoint) / (config.farPoint - config.nearPoint);
    float polynomialTerm = EvaluatePolynomial(normalizedDepth, config.polynomialDegree, config.polynomialWeight);
    float skewedDepth = pow(normalizedDepth, config.skewFactor);
    float transitionFactor = smoothstep(0.0f, config.transitionWidth, (depth - config.transitionPoint) / (config.farPoint - config.transitionPoint));
    float linearTerm = slerp(normalizedDepth * config.nearSlope, normalizedDepth * config.farSlope, transitionFactor);
    float mappedDepth = slerp(linearTerm, polynomialTerm, transitionFactor);
    return config.nearPoint + (mappedDepth + config.offsetFactor) * (config.farPoint - config.nearPoint);
}

inline float3 RGBtoHSL(float3 color)
{
    float maxComponent = max(color.r, max(color.g, color.b));
    float minComponent = min(color.r, min(color.g, color.b));
    float range = maxComponent - minComponent;

    float3 hsl;
    hsl.z = (maxComponent + minComponent) / 2.0; // Lightness

    if (range == 0.0)
    {
        hsl.x = 0.0; // Hue is undefined 
        hsl.y = 0.0; // Saturation
    }
    else
    {
        hsl.y = range / (1.0 - abs(2.0 * hsl.z - 1.0)); // Saturation

        float delta = range / (6.0 * maxComponent);

        if (color.r == maxComponent)
        {
            hsl.x = (color.g - color.b) * delta;
        }
        else if (color.g == maxComponent)
        {
            hsl.x = (2.0 + (color.b - color.r) * delta);
        }
        else
        {
            hsl.x = (4.0 + (color.r - color.g) * delta);
        }

        hsl.x = fmod(hsl.x, 1.0); // Wrap hue to [0,1] range
    }

    return hsl;
}

inline float3 HSLtoRGB(float3 hsl)
{
    float C = (1.0 - abs(2.0 * hsl.z - 1.0)) * hsl.y; // Chroma
    float x = C * (1.0 - abs(fmod(hsl.x * 6.0, 2.0) - 1.0));
    float m = hsl.z - C / 2.0; // Lightness offset

    float3 rgbPrime;

    if (hsl.x < 1.0 / 6.0)
    {
        rgbPrime = float3(C, x, 0);
    }
    else if (hsl.x < 2.0 / 6.0)
    {
        rgbPrime = float3(x, C, 0);
    }
    else if (hsl.x < 3.0 / 6.0)
    {
        rgbPrime = float3(0, C, x);
    }
    else if (hsl.x < 4.0 / 6.0)
    {
        rgbPrime = float3(0, x, C);
    }
    else if (hsl.x < 5.0 / 6.0)
    {
        rgbPrime = float3(x, 0, C);
    }
    else
    {
        rgbPrime = float3(C, 0, x);
    }

    return rgbPrime + m;
}

        totalCount[0]++;

    }
    if (color2[1] > 0)
    {

        totalCount[1]++;
    }
    if (color3[1] > 0)
    {
        ret[1] += color3[1];
        totalCount[1]++;
    }

    if (color1[2] > 0)
    {
        ret[2] += color1[2];
        totalCount[2]++;

    {
        ret[2] += color2[2];

    }

    }

    {
        ret[3] += color1[3];
        totalCount[3]++;
    }

float DepthCurve(float depth)
{
    // Define the depth curve parameters
    float curveExponent = 1.5; // Exponent value
    float curveOffset = 0.01; // Offset value
    float curveSteepness = 2.0; // Steepness value
    float curveRange = .80; // Range value

    // Apply the depth curve
    float depthValue = (pow(depth, curveExponent) * curveSteepness) / (1.0 + curveSteepness * (depth - curveOffset)) * curveRange;

    // Return the final depth value
    return depthValue;
}
float InverseDepthCurve(float depthValue)
{
    const float curveExponent = 1.5;
    const float curveOffset = 0.01;
    const float curveSteepness = 2.0;
    const float curveRange = 0.80;

    float numerator = curveSteepness * pow(depthValue * curveRange, 1.0f / curveExponent); // Simplified
    float denominator = 1.0 + curveSteepness * (curveOffset - depthValue);
    return numerator / denominator;
}

// Generate a Gaussian kernel
void GenerateGaussianKernel(out float kernel[9], float sigma)
{
    float sum = 0.0;
    int radius = 4; // Kernel size is 2*radius+1 = 9

    for (int i = -radius; i <= radius; ++i)
    {
        kernel[i + radius] = Gaussian(float(i), sigma);
        sum += kernel[i + radius];
    }

    // Normalize the kernel
    for (i = 0; i < 9; ++i)
    {
        kernel[i] /= sum;
    }
}

// Gaussian depth texture sampling function
float GaussianDepthSample(float2 oosz, Texture2D<float> tex, float2 uv)
{
    float kernel[9];
    GenerateGaussianKernel(kernel, 2);
    float sigma = 2;
    float depth = 0.0;
    float weightSum = 0.0;
    float radius = 4;
    // Accumulate samples
    for (int x = -radius; x <= radius; ++x)
    {
        for (int y = -radius; y <= radius; ++y)
        {
            float2 offset = float2(x, y) * oosz;
            float sampleDepth = ClampEpsilon(tex.SampleLevel(sampleTypeMirror, uv + offset, 0).r) * DepthScale;
            float weight = kernel[x + radius] * kernel[y + radius];
            depth += sampleDepth * weight;
            weightSum += weight;
        }
    }

    // Normalize the depth value
    return depth / weightSum;
}



// Select sigmoid steepness based on gradient magnitude
float SelectSigmoidSteepness(float averageGradientMagnitude, float2 steepnessMinMax)
{
    float normalizedMagnitude = saturate(averageGradientMagnitude);
    return lerp(steepnessMinMax.x, steepnessMinMax.y, normalizedMagnitude);
}

// Adjust steepness dynamically based on gradient magnitude
float AdjustSteepness(float steepness, float gradientMagnitude, float threshold, float maxAdjustment)
{
    if (gradientMagnitude > threshold)
    {
        float adjustmentFactor = 1.0 - min((gradientMagnitude - threshold) / maxAdjustment, 1.0);
        return steepness * adjustmentFactor;
    }
    return steepness;
}

// Compute adjusted depth curve
float ComputeAdjustedDepthCurve(float depth, float steepness)
{
    return pow(depth, 1.0 / (1.0 + steepness));
}

// Function to calculate the adjusted depth using physics-based principles
inline float ComputeAdjustedDepth(float depth, float2 gradient)
{
    float2 steepnessMinMax = vec2(-.1, .1);
    //AverageGradientMagnitude avgGradMag = ComputeAverageGradientMagnitude(config.oosz, depthMap, uv, radius, sigma);
    float steepness = SelectSigmoidSteepness((gradient.x + gradient.y) / 2, steepnessMinMax);
    float adjustedSteepness = AdjustSteepness(steepness, length(gradient), steepnessMinMax.x, steepnessMinMax.y);
    return ComputeAdjustedDepthCurve(depth, adjustedSteepness);
}

inline float2 GetGradient(float2 uv)
{
    float depth = ClampEpsilon(depthMap.SampleLevel(sampleTypeMirror, uv, 0).r) * DepthScale;
    return vec2(ddx_fine(depth), ddy_fine(depth));
}

vec4 CrossLRUDf(vec2 oosz, Texture2D<float> tex, vec2 uv, float range)
{
    return vec4(
        ClampEpsilon(tex.SampleLevel(sampleTypeMirror, uv + oosz * vec2(-range, 0), 0).r),
        ClampEpsilon(tex.SampleLevel(sampleTypeMirror, uv + oosz * vec2(range, 0), 0).r),
        ClampEpsilon(tex.SampleLevel(sampleTypeMirror, uv + oosz * vec2(0, -range), 0).r),
        ClampEpsilon(tex.SampleLevel(sampleTypeMirror, uv + oosz * vec2(0, range), 0).r)
    ) * DepthScale;
}

inline float dep2D(float2 oosz, Texture2D<float> depthMap, float2 uv)
{
    float2 gradient = GetGradient(uv);
    float depthVariance = variation(ClampEpsilon(CrossLRUDf(oosz, depthMap, uv, normalRadius)));
    float bilateralWeight = exp(-0.5 * ClampEpsilon(dot(gradient, gradient)) / ClampEpsilon(depthVariance));
    return slerp(ClampEpsilon(depthMap.SampleLevel(sampleTypeMirror, uv, 0).r), ClampEpsilon(dot(gradient, config.gradient)), ClampEpsilon(bilateralWeight * .1)) * DepthScale;
}


inline float dep2D(Texture2D<float> depthMap, float2 uv)
{
    return dep2D(ooszDep(), depthMap, uv);
}

inline float dep2D(float2 uv)
{
    return dep2D(depthMap, uv);
}

// Convert CIE XYZ to linear sRGB 
float3 XYZtoRGB(float3 XYZ)
{
    const float3x3 XYZ_TO_RGB = float3x3(
        3.2404542, -1.5371385, -0.4985314,
        -0.9692660, 1.8760108, 0.0415560,
        0.0556434, -0.2040259, 1.0572252
    );
    return max(float3(0, 0, 0), mul(XYZ_TO_RGB, XYZ)); // Clip to ensure non-negative values
}


float3 GetColorFromWavelength(float wavelength, float3 wavelengthColorData[200])
{
    // Use a cubic spline interpolation to find the corresponding color value
    int index = (int) ((wavelength - 0.6328) / 0.1) % 200;
    float t = (wavelength - 0.6328) / 0.1 - index;
    float3 c0 = wavelengthColorData[index];
    float3 c1 = wavelengthColorData[index + 1];
    float3 c2 = wavelengthColorData[index + 2];
    float3 c3 = wavelengthColorData[index + 3];
    return c0 * (-0.5 * t + 0.5 * t * t - 0.5 * t * t * t) +
           c1 * (1.0 + t + 0.5 * t * t - 1.5 * t * t * t) +
           c2 * (-0.5 * t + 1.5 * t * t + 0.5 * t * t * t) +
           c3 * (0.5 * t * t - 0.5 * t * t * t);
}
float3 WavelengthToRGB_Spectral(float wavelength, Texture2D<float> spectralPowerDistribution)
{
    // Clamp wavelength to valid range
    wavelength = clamp(wavelength, 380.0, 780.0);

    // Sample the spectral power distribution across the visible spectrum
    float3 XYZ = float3(0.0, 0.0, 0.0);
    for (int i = 0; i < 81; ++i)
    {
        float lambda = 380.0 + i * 5.0; // Wavelength in nm
        float2 uv = float2(lambda / 780.0, 0.5); // Map to [0,1] range
        float power = dep2D(GetOosz(spectralPowerDistribution), spectralPowerDistribution, uv);

        XYZ += XYZColorMatchingFunctions[i] * power;
    }

    // Convert XYZ to linear sRGB, scale, and clamp
    float3 linearRGB = max(float3(0, 0, 0), XYZtoRGB(XYZ));

    // Optional: Gamma correction for display (if not done elsewhere)
    // float3 sRGB = pow(linearRGB, 1.0 / 2.2);
    return saturate(linearRGB);
}

float4 WavelengthSpectrumPS(Texture2D<float4> diffuse, Texture2D<float3> normal, Texture2D<float> depth,
    float2 uv)
{
    // Sample the diffraction pattern texture
    float4 diffraction = tex2D(diffuse, uv);

    // Calculate the wavelength from the texture coordinates
    float wavelength = 0.6328 + (uv.x * 0.1); // 632.8nm to 732.8nm spectrum

    // Use a high-accuracy wavelength-color mapping function
    float3 color = mir2D(diffuse, uv).xyz;

    // Apply a spectral power distribution correction
    color *= GetSPDCorrection(wavelength);

    // Modulate the color with the diffraction pattern
    color *= diffraction.rgb;

    return float4(color, 1);
}



float3 GaussianSample(float2 oosz, Texture2D<float4> tex, float2 uv, Texture2D<float> depthMap, float focusDepth)
{
    float kernel[9];
    GenerateGaussianKernel(kernel, 2);
    float sigma = 2;
    float3 color = 0.0;
    float weightSum = 0.0;
    float radius = 4;
    float focus = abs(focusDepth - dep2D(uv));
    if (focus < .2)
    {
        return mir2D(tex, uv).xyz;
    }
    else if (focus < .3)
    {
        radius = 1;
    }
    else if (focus < .5)
    {
        radius = 2;
    }
    else if (focus < .7)
    {
        radius = 3;
    }
    
    [unroll]
    for (int x = -radius; x <= radius; ++x)
    {
        [unroll]
        for (int y = -radius; y <= radius; ++y)
        {
            float2 offset = float2(x, y) * oosz;
            float3 sampleColor = mir2D(tex, uv + offset).xyz;
            float depth = dep2D(depthMap, uv + offset);
            float focus = length((uv + float2(x, y) * oosz) - .5);
            float weight = kernel[x + radius] * kernel[y + radius] * focus;
            color = color + sampleColor * weight;
            weightSum += weight;
        }
    }
    
    return slerp(mir2D(tex, uv).rgb, color / weightSum, length(uv - .5) * focus);
}



float2 HeatHaze(float2 oosz, Texture2D<float> tex, float2 uv, float time, float xMul, float yMul, float timeMul, float finalMul)
{
    return (cos(uv.x * xMul * time * timeMul) + sin(uv.y * yMul * time * timeMul)) / 4 * finalMul * dep2D(tex, uv) * oosz;
}

float3 HeatHaze(float2 oosz, Texture2D tex, float2 uv, float time, float xMul, float yMul, float timeMul, float finalMul)
{
    return mir2D(tex, uv + HeatHaze(oosz, tex, uv, time, xMul, yMul, timeMul, finalMul).xy).rgb;
}


// Helper function to map a value from one range to another with scaling and offset
inline float remap(float value, float inMin, float inMax, float outMin, float outMax, float outOffset, float outScale)
{
    float remapped = outMin + (value - inMin) * (outMax - outMin) / (inMax - inMin);
    return (remapped + outOffset) * outScale;
}
inline float4 remap(float4 value, float4 inMin, float4 inMax, float4 outMin, float4 outMax, float4 outOffset, float4 outScale)
{
    float4 remapped = outMin + (value - inMin) * (outMax - outMin) / (inMax - inMin);
    return (remapped + outOffset) * outScale;
}
// Logistic function (centered around 0, range [-1, 1])
inline float logistic(float x, float steepness)
{
    return 2.0f / (1.0f + exp(-2.0f * steepness * x)) - 1.0f;
}
inline float4 logistic(float4 x, float4 steepness)
{
    return 2.0f / (1.0f + exp(-2.0f * steepness * x)) - 1.0f;
}
inline float3 logistic(float3 x, float3 steepness)
{
    return 2.0f / (1.0f + exp(-2.0f * steepness * x)) - 1.0f;
}
inline float2 logistic(float2 x, float2 steepness)
{
    return 2.0f / (1.0f + exp(-2.0f * steepness * x)) - 1.0f;
}

inline float sigmoid1(float x, float steepness, float nearValue, float farValue, float offset, float scale)
{
    float normalized = logistic(x, steepness);
    return remap(normalized, SIGMOID_NEAR, SIGMOID_FAR, nearValue, farValue, offset, scale);
}

inline float4 sigmoid4(float4 x, float4 steepness, float4 nearValue, float4 farValue, float4 offset, float4 scale)
{
    float4 normalized = logistic(x, steepness);
    return remap(normalized, SIGMOID_NEAR, SIGMOID_FAR, nearValue, farValue, offset, scale);
}

inline float sigmoid1(float x, float steepness, float offset, float scale)
{
    return sigmoid1(x, steepness, SIGMOID_NEAR, SIGMOID_FAR, offset, scale);
}
inline float2 sigmoid2(float2 x, float2 steepness, float2 offset, float2 scale)
{
    return float2(sigmoid1(x.x, steepness.x, SIGMOID_NEAR, SIGMOID_FAR, offset.x, scale.x),
        sigmoid1(x.y, steepness.y, SIGMOID_NEAR, SIGMOID_FAR, offset.y, scale.y));
}

inline float sigmoid(float x, float steepness, float nearValue, float farValue, float offset, float scale)
{
    float normalized = logistic(x, steepness);
    return remap(normalized, 0.0f, 1.0f, nearValue, farValue, offset, scale);
}

inline float4 sigmoid(float4 x, float4 steepness, float4 nearValue, float4 farValue, float4 offset, float4 scale)
{
    float4 normalized = logistic(x, steepness);
    return remap(normalized, .0f, 1.0f, nearValue, farValue, offset, scale);
}

// Sigmoid function
inline float sigmoid(float x, float slope, float nearRange, float farRange)
{
    return 1.0 / (1.0 + exp(-slope * (x - nearRange))) - 1.0 / (1.0 + exp(-slope * (x - farRange)));
}

inline float2 GetModulation(Texture2D<float> depthMap, float2 uv)
{
    // Sample the depth map
    float depth = dep2D(depthMap, uv);

    // Calculate the depth gradient
    float2 depthGradient = float2(ddx(depth), ddy(depth));

    float nearSlope = -0.1;
    float farSlope = 0.1;
    float nearRange = 0.2;
    float farRange = 0.8;

    // Calculate the sigmoid slope based on the depth gradient
    float sigmoidSlopeX = lerp(nearSlope, farSlope, saturate(depthGradient.x));
    float sigmoidSlopeY = lerp(nearSlope, farSlope, saturate(depthGradient.y));

    // Apply the sigmoid function to the depth gradient components
    float2 sigmoidCurve = float2(
        sigmoid(saturate(depthGradient.x), sigmoidSlopeX, nearRange, farRange),
        sigmoid(saturate(depthGradient.y), sigmoidSlopeY, nearRange, farRange)
    );

    // Clamp the output to a small epsilon value for stability
    sigmoidCurve = clamp(sigmoidCurve, -0.2, 0.2);

    return sigmoidCurve;
}




float XCrossLRUDfAvg(float2 oosz, Texture2D<float> tex, float2 uv, float range)
{
    range = min(4, max(1, range));
    float ret = 0;
    float weightTotal = 0;
    for (int x = -range; x <= range; x += 2)
    {
        for (int y = -range; y <= range; y += 2)
        {
            // Calculate weight based on distance from center pixel
            float weight = 1.0 / (1.0 + sqrt((x * oosz.x) * (x * oosz.x) + (y * oosz.y) * (y * oosz.y)));
            // Check neighboring pixels within the range
            float depth = dep2D(oosz, tex, uv + vec2(x, y) * oosz);
// Accumulate the weighted depth value
            ret += depth * weight;
            weightTotal += weight;
            
        }
    }
// Normalize by total weight
    ret = (ret / weightTotal);
    return ret;
}

float3 GetNormal2(float2 oosz, Texture2D<float> depthMap, float2 uv, float dist, float2 modulation)
{
    float centerDepth = dep2D(oosz, depthMap, uv);
    float2 tex1 = uv - float2(oosz.x, 0) * dist;
    float2 tex2 = uv - float2(0.0, oosz.y) * dist;
    float2 tex3 = uv + float2(oosz.x, 0) * dist;
    float2 tex4 = uv + float2(0.0, oosz.y) * dist;
    
    float xDepth = dep2D(oosz, depthMap, tex1);
    float yDepth = dep2D(oosz, depthMap, tex2);

    // Compute view-space positions
    float3 centerPos = float3(uv, centerDepth);
    float3 xPos = float3(tex1, xDepth);
    float3 yPos = float3(tex2, yDepth);

    // Compute view-space normal
    float3 viewNormal = (cross(normalize(yPos - centerPos), normalize(xPos - centerPos)));
    return normalize(viewNormal * vec3(1, 1, -1));
}

vec2 GetGradientBetween(vec2 oosz, vec2 uv1, vec2 uv2)
{
    vec2 dir = uv2 - uv1;
    float pixelDistance = length(dir / oosz); // Distance in pixels
    
    // Optimize sampling based on pixel distance
    const float minPixelDistance = 0.5;
    int sampleCount = max(1, min(10, int(ceil(pixelDistance / minPixelDistance))));

    vec2 gradientSum = vec2(0.0, 0.0);
    float totalDepthChange = 0.0;
    float prevDepth = dep2D(oosz, depthMap, uv1);

    for (int i = 1; i <= sampleCount; ++i)
    {
        float t = float(i) / float(sampleCount);
        vec2 sampleUV = slerp(uv1, uv2, t);
        float currentDepth = dep2D(oosz, depthMap, sampleUV);

        // Calculate depth change and accumulate
        float depthChange = currentDepth - prevDepth;
        totalDepthChange += depthChange;

        // Weighted gradient contribution
        gradientSum += depthChange * (dir / pixelDistance);

        prevDepth = currentDepth;
    }

    // Normalize the gradient by the total depth change
    if (totalDepthChange != 0.0)
    {
        gradientSum /= totalDepthChange;
    }

    return gradientSum;
}

float3 HueToRGB(float hue)
{
    float r = abs(hue * 6.0 - 3.0) - 1.0;
    float g = 2.0 - abs(hue * 6.0 - 2.0);
    float b = 2.0 - abs(hue * 6.0 - 4.0);
    return saturate(float3(r, g, b));
}
float RGBtoHue(float3 rgb)
{
    float M = max(max(rgb.r, rgb.g), rgb.b);
    float m = min(min(rgb.r, rgb.g), rgb.b);
    float C = M - m; // Chroma

    float hue = 0.0; // Initialize hue

    if (C != 0)
    {
        if (M == rgb.r)
        {
            hue = fmod(((rgb.g - rgb.b) / C), 6.0);
        }
        else if (M == rgb.g)
        {
            hue = ((rgb.b - rgb.r) / C) + 2.0;
        }
        else
        {
            hue = ((rgb.r - rgb.g) / C) + 4.0;
        }
        hue *= 60.0;
        if (hue < 0.0)
            hue += 360.0;
    }

    return hue / 360.0; // Normalize to [0, 1]
}


float ApplyGammaCorrection(float colorComponent)
{
    if (colorComponent > 0.04045f)
    {
        return pow((colorComponent + 0.055f) / 1.055f, 2.4f);
    }
    else
    {
        return colorComponent / 12.92f;
    }
}


float3 CreateRainbowSpectrum(float2 uv, float3 viewDir)
{
    // Adjust the hue based on the viewing direction
    float hueShift = atan2(viewDir.y, viewDir.x) / (2.0 * PI);

    // Calculate the base hue from the uv coordinates
    float baseHue = uv.x + uv.y;

    // Combine the base hue with the directional hue shift
    float finalHue = frac(baseHue + hueShift);

    // Convert the final hue to an RGB color
    return HueToRGB(finalHue);
}

float3 rgb2hsv_eps(float3 c)
{
    float4 K = float4(0.0, -1.0 / 3.0, 2.0 / 3.0, -1.0);
    float4 p = slerp(float4(c.bg, K.wz), float4(c.gb, K.xy), step(c.b, c.g));
    float4 q = slerp(float4(p.xyw, c.r), float4(c.r, p.yzx), step(p.x, c.r));
    float d = q.x - min(q.w, q.y);
    float e = 1.0e-10;
    float3 hsv = float3(abs(q.z + (q.w - q.y) / (6.0 * d + e)), d / (q.x + e + 1.0e-6), q.x);

    // Add a smoothstep function to gradually transition to black
    float lum = hsv.z * 2.0 - 1.0; // Map value to -1..1 range
    lum = smoothstep(-1.0, 0.0, lum); // Gradually transition to black
    hsv.z = lum * 0.5 + 0.5; // Map back to 0..1 range

    return hsv;
}

float3 hsv2rgb_eps(float3 c)
{
    float4 K = float4(1.0, 2.0 / 3.0, 1.0 / 3.0, 3.0);
    float3 p = abs(frac(c.xxx + K.xyz) * 6.0 - K.www);
    float3 rgb = c.z * slerp(K.xxx, clamp(p - K.xxx, 0.0, 1.0), c.y + 1.0e-6);

    // Add a smoothstep function to gradually transition to black
    float lum = rgb.x * 2.0 - 1.0; // Map value to -1..1 range
    lum = smoothstep(-1.0, 0.0, lum); // Gradually transition to black
    rgb.x = lum * 0.5 + 0.5; // Map back to 0..1 range

    return rgb;
}

float3 rgb2hsv(float3 c)
{
    float4 K = float4(0.0, -1.0 / 3.0, 2.0 / 3.0, -1.0);
    float4 p = slerp(float4(c.bg, K.wz), float4(c.gb, K.xy), step(c.b, c.g));
    float4 q = slerp(float4(p.xyw, c.r), float4(c.r, p.yzx), step(p.x, c.r));
    float d = q.x - min(q.w, q.y);
    float e = 1.0e-10;
    float3 hsv = float3(abs(q.z + (q.w - q.y) / (6.0 * d + e)), d / (q.x + e + 1.0e-6), q.x);

    // Add a smoothstep function to gradually transition to black
    float lum = hsv.z * 2.0 - 1.0; // Map value to -1..1 range
    lum = smoothstep(-1.0, 0.0, lum); // Gradually transition to black
    hsv.z = lum * 0.5 + 0.5; // Map back to 0..1 range

    return hsv;
}

float3 hsv2rgb(float3 c)
{
    float4 K = float4(1.0, 2.0 / 3.0, 1.0 / 3.0, 3.0);
    float3 p = abs(frac(c.xxx + K.xyz) * 6.0 - K.www);
    float3 rgb = c.z * slerp(K.xxx, clamp(p - K.xxx, 0.0, 1.0), c.y + 1.0e-6);

    // Add a smoothstep function to gradually transition to black
    float lum = rgb.x * 2.0 - 1.0; // Map value to -1..1 range
    lum = smoothstep(-1.0, 0.0, lum); // Gradually transition to black
    rgb.x = lum * 0.5 + 0.5; // Map back to 0..1 range

    // Add a smoothstep function to gradually transition to white
    lum = rgb.x * 2.0 - 1.0; // Map value to -1..1 range
    lum = smoothstep(0.0, 1.0, lum); // Gradually transition to white
    rgb.x = lum * 0.5 + 0.5; // Map back to 0..1 range

    return rgb;
}

float4 HSVtoRGB(float4 hsv)
{
    float h = hsv.x;
    float s = hsv.y;
    float v = hsv.z;

    if (s == 0.0f)
        return float4(v, v, v, 1.0);

    h /= 60.0f;
    int i = int(floor(h));
    float f = h - float(i);
    float p = v * (1.0f - s);
    float q = v * (1.0f - s * f);
    float t = v * (1.0f - s * (1.0f - f));

    if (i == 0)
        return vec4(v, t, p, 1.0);
    if (i == 1)
        return vec4(q, v, p, 1.0);
    if (i == 2)
        return vec4(p, v, t, 1.0);
    if (i == 3)
        return vec4(p, q, v, 1.0);
    if (i == 4)
        return vec4(t, p, v, 1.0);

    return vec4(v, p, q, 1.0);
}

vec3 RGBtoHSV(vec3 rgb)
{
    float R = rgb.r;
    float G = rgb.g;
    float B = rgb.b;
    float maxC = max(R, max(G, B));
    float minC = min(R, min(G, B));
    float delta = maxC - minC;

    float H = 0;
    if (delta == 0)
    {
        H = 0;
    }
    else if (maxC == R)
    {
        H = 60 * (fmod(((G - B) / delta), 6));
    }
    else if (maxC == G)
    {
        H = 60 * (((B - R) / delta) + 2);
    }
    else if (maxC == B)
    {
        H = 60 * (((R - G) / delta) + 4);
    }

    float S = (maxC == 0) ? 0 : (delta / maxC);
    float V = maxC;

    return float3(H, S, V);
}

inline float Luminance(float3 color)
{
    return dot(color, float3(0.2126, 0.7152, 0.0722)); // Standard luminance calculation from RGB
}


float3 lerpSigmoidLuminocity(float3 baseColor, float3 blendColor1, float3 blendColor2, float alpha, float beta)
{
    float baseLuminance = Luminance(baseColor);
    float blendLuminance1 = Luminance(blendColor1);
    float blendLuminance2 = Luminance(blendColor2);

    // Advanced sigmoid for non-linear mix ratio
    float mixRatio = sigmoid((blendLuminance1 - blendLuminance2) / (1.0 + abs(blendLuminance1 - blendLuminance2)), alpha, beta, 1);
    float3 mixedColor = slerp(blendColor1, blendColor2, mixRatio);

    // Adjust luminance non-linearly
    float mixedLuminance = Luminance(mixedColor);
    float luminanceAdjustment = sigmoid(baseLuminance / mixedLuminance, alpha, beta, 1);
    mixedColor *= luminanceAdjustment;

    return mixedColor;
}


float3 HueShift(float3 color, float hue)
{
    // Use a rotation matrix to shift the hue of the color
    const float3 k = float3(0.57735, 0.57735, 0.57735); // 1 / sqrt(3)
    float cosAngle = cos(hue);
    float sinAngle = sin(hue);
    return color * cosAngle + cross(k, color) * sinAngle + k * dot(k, color) * (1.0 - cosAngle);
}


float RefractionIndex(float interactionStrength)
{
    return 1.0 + sigmoid(interactionStrength, 0.5, 15.0, 1) * 0.3; // Adjust refraction index
}


float3 FresnelSchlick(float3 F0, float cosTheta, float3 fresnelPower)
{
    F0 = ClampEpsilon11(F0);
    return F0 + (1.0 - F0) * pow(ClampEpsilon(1.0 - cosTheta), fresnelPower);
}

float3 FresnelSchlick(float3 F0, float3 H, float3 N, float3 power)
{
    return FresnelSchlick(F0, dot(H, N), power);
}


float FresnelSchlick(float refractiveIndex, float surroundingRefractiveIndex, float cosIncidenceAngle, float power)
{
    float F0 = (refractiveIndex * refractiveIndex - surroundingRefractiveIndex * surroundingRefractiveIndex) / (refractiveIndex * refractiveIndex + surroundingRefractiveIndex * surroundingRefractiveIndex);
    return F0 + (1 - F0) * pow(1 - cosIncidenceAngle, power);
}

vec4 Fresnel(vec4 cosTheta)
{
    return exp2((cosTheta * -5.55473 - 6.98316) * cosTheta);
}
vec4 Schlick(vec4 cosTheta, float4 fresnelPower)
{
    return Fresnel(cosTheta) / (cosTheta * (1 - fresnelPower)) * fresnelPower;
}

vec4 Schlick2(vec4 F0, vec4 cosTheta, float4 fresnelPower)
{
    return F0 + (1.0 - F0) * pow(ClampEpsilon(1.0 - cosTheta), fresnelPower);
}


vec3 ComputeFresnel(vec3 rayDirection, vec3 normal, vec3 refractiveIndex)
{
    float cosTheta = dot(-rayDirection, normal);
    float3 r0 = (1.0 - refractiveIndex) / (1.0 + refractiveIndex);
    r0 = r0 * r0;
    return r0 + (1.0 - r0) * pow(ClampEpsilon(1.0 - cosTheta), FresnelPower);
}



vec4 PhaseFunction(vec4 normal, vec4 viewDir, vec4 phaseFactor)
{
    float cosTheta = ClampEpsilon(dot(normal.xyz, viewDir.xyz));
    return (1.0f / (4.0f * 3.14159f) * (1.0f + (phaseFactor * cosTheta * cosTheta)));
}

float4 RotateAroundAxis(float4 v, float4 axis, float angle)
{
    // Normalize the axis
    axis = normalize(axis);

    // Compute the cosine and sine of the angle
    float c = cos(angle);
    float s = sin(angle);

    // Compute the rotation matrix
    float4x3 R;
    R[0][0] = c + (1.0 - c) * axis.x * axis.x;
    R[0][1] = (1.0 - c) * axis.x * axis.y - s * axis.z;
    R[0][2] = (1.0 - c) * axis.x * axis.z + s * axis.y;
    
    R[1][0] = (1.0 - c) * axis.y * axis.x + s * axis.z;
    R[1][1] = c + (1.0 - c) * axis.y * axis.y;
    R[1][2] = (1.0 - c) * axis.y * axis.z - s * axis.x;
   
    R[2][0] = (1.0 - c) * axis.z * axis.x - s * axis.y;
    R[2][1] = (1.0 - c) * axis.z * axis.y + s * axis.x;
    R[2][2] = c + (1.0 - c) * axis.z * axis.z;
   
    // Multiply the vector by the rotation matrix
    return float4(mul(R, v.xyz).xyz, v.w);
}

float4 RainbowOilFilm(float2 uv, float3 normal, float3 lightDir, float3 viewDir, float4 color)
{
    color = ClampEpsilon(color);

    // The thickness of the oil film in nanometers
    float thickness = 200.0 + 100.0 * sin(uv.x * 10.0) * cos(uv.y * 10.0);
    
    // The refractive indices of the oil and air
    float n1 = 1.33; // oil
    float n2 = 1.00; // air
    
    // The angle of incidence between the normal and the light direction
    float theta = acos(dot(normal, lightDir));
    
    // The angle of refraction using Snell's law
    float phi = asin(n2 * sin(theta) / n1);
    
    // The reflectance of the oil film using Fresnel's equations
    float Rs = pow((n1 * cos(theta) - n2 * cos(phi)) / (n1 * cos(theta) + n2 * cos(phi)), 2.0);
    float Rp = pow((n1 * cos(phi) - n2 * cos(theta)) / (n1 * cos(phi) + n2 * cos(theta)), 2.0);
    float R = (Rs + Rp) / 2.0;
    
    // The phase difference between the reflected rays using the thin film interference formula
    float delta = 4.0 * PI * n1 * thickness * cos(phi) / 550.0; // 550 nm is the wavelength of green light
    
    // The interference color using the RGB color model
    float Ir = (R + R * cos(delta - 4.0 * PI / 3.0));
    float Ig = (R + R * cos(delta));
    float Ib = (R + R * cos(delta + 4.0 * PI / 3.0));
    
    // The final color after adding the interference color to the original color
    float4 result;
    result.r = (color.r + Ir);
    result.g = (color.g + Ig);
    result.b = (color.b + Ib);
    result.a = color.a;
    
    // Return the final color
    return result;
}

// Function to sample the texture and apply modulation
float4 SampleWithModulation(Texture2D tex, float2 uv, float2 offset, float2 oosz, float2 gradientModulation)
{
    return mir2D(tex, uv + offset * oosz * gradientModulation);
}

// Simplified blur function
float4 Blur(float2 oosz, Texture2D tex, float2 uv, float dist, float strength, float2 gradientModulation)
{
    // Sample the original texture
    float4 color = mir2D(tex, uv);
    float4 blurColor[4];
    
    // Calculate blur samples
    blurColor[0] = SampleWithModulation(tex, uv, float2(-dist, -dist), oosz, gradientModulation);
    blurColor[1] = SampleWithModulation(tex, uv, float2(dist, dist), oosz, gradientModulation);
    blurColor[2] = SampleWithModulation(tex, uv, float2(-dist, dist), oosz, gradientModulation);
    blurColor[3] = SampleWithModulation(tex, uv, float2(dist, -dist), oosz, gradientModulation);
    
    // Blend the blur samples
    float4 blendedBlur = slerp(blurColor[3], slerp(blurColor[2], slerp(blurColor[1], blurColor[0], 0.35), 0.35), 0.35);
    
    // Combine with original color
    return ClampEpsilon11(slerp(color, blendedBlur, strength));
}

float CalculateRefractiveIndexUsingSellmeier(MaterialProperties material, float wavelength)
{
    // Convert wavelength to micrometers (if needed)
    // Assuming wavelength is given in nanometers, convert to micrometers
    float lambdaMicrometers = wavelength * 1e-3;
    float lambdaSquared = lambdaMicrometers * lambdaMicrometers;
    
    // Initialize nSquared with 1.0 (accounting for vacuum contribution)
    float nSquared = 1.0;

    // Calculate the refractive index using the Sellmeier equation
    for (int i = 0; i < 3; i++)
    {
        float B = material.sellmeierCoefficients[i].B;
        float C = material.sellmeierCoefficients[i].C;
        // Ensure no division by zero or negative value under the square root
        if (lambdaSquared != C)
        {
            nSquared += (B * lambdaSquared) / (lambdaSquared - C);
        }
    }

    // Return the square root of nSquared to get the refractive index
    return sqrt(nSquared);
}

// Calculates the geometric attenuation factor
float GGXGeometry(float NdotV, float alphaRoughness, float NdotL)
{
    float k = (alphaRoughness * alphaRoughness) * 0.5;
    float G1V = NdotV / (NdotV * (1.0 - k) + k);
    float G1L = NdotL / (NdotL * (1.0 - k) + k);
    return (G1V * G1L);
}

// Calculates the microfacet distribution
float GGXDistribution(float NdotH, float alphaRoughness)
{
    float alphaRoughness2 = alphaRoughness * alphaRoughness;
    float NdotH2 = NdotH * NdotH;
    float denom = NdotH2 * (alphaRoughness2 - 1.0) + 1.0;
    return alphaRoughness2 / (PI * denom * denom);
}

float GeometrySmith(float3 N, float3 V, float3 L, float roughness)
{
    float NdotV = ClampEpsilon(dot(N, V));
    float NdotL = ClampEpsilon(dot(N, L));
    float k = roughness * roughness * 0.5;
    float G1V = NdotV / (NdotV * (1.0 - k) + k);
    float G1L = NdotL / (NdotL * (1.0 - k) + k);
    return G1V * G1L;
}

float4 SpecularTerm(float3 normal, float3 viewDir, float3 lightDir, float4 specularPower, float4 specularIntensity)
{
    float3 reflectDir = reflect(-ClampEpsilon11(lightDir), ClampEpsilon11(normal));
    return pow(ClampEpsilon(dot(reflectDir, ClampEpsilon11(viewDir))), specularPower) * specularIntensity;
}


float ggxDistribution(float NdotH, float roughness)
{
    float alphaRoughness = roughness * roughness;
    float NdotH2 = NdotH * NdotH;
    float denom = NdotH2 * (alphaRoughness - 1.0) + 1.0;
    return alphaRoughness / (PI * denom * denom);
}

// GGX Geometry
float ggxGeometry(float NdotV, float NdotL, float roughness)
{
    float alphaRoughness = roughness * roughness;
    float k = alphaRoughness * 0.5;
    float G1V = NdotV / (NdotV * (1.0 - k) + k);
    float G1L = NdotL / (NdotL * (1.0 - k) + k);
    return G1V * G1L;
}

float SmithGGXCorrelated(float NdotL, float NdotV, float roughness2)
{
    roughness2 = ClampEpsilon(roughness2);
    float a2 = ClampEpsilon(roughness2 * roughness2);
    float sqrtNdotV = sqrt(ClampEpsilon(NdotV));
    float sqrtNdotL = sqrt(ClampEpsilon(NdotL));
    float GGXV = NdotL * (sqrtNdotV * sqrtNdotV * (1.0 - a2) + a2);
    float GGXL = NdotV * (sqrtNdotL * sqrtNdotL * (1.0 - a2) + a2);
    return 0.5 / ClampEpsilon(GGXV + GGXL);
}

float3 FresnelSchlickDiffuse(float NdotL, float3 F0)
{
    return ClampEpsilon((1.0 - F0) * (1.0 - pow(ClampEpsilon(1.0 - NdotL), FresnelPower)));
}

{
    return ClampEpsilon(F0 + (1.0 - F0) * pow(ClampEpsilon(1.0 - VdotH), FresnelReflectance));
}

float3 FresnelReflection(float3 F0, float VdotH)
{
    return ClampEpsilon(F0 + (1.0 - F0) * pow(ClampEpsilon(1.0 - VdotH), FresnelReflectance));
}
// Improved Fresnel Schlick for Dielectrics (Non-Metals)
float3 FresnelSchlickDielectric(float cosTheta, float3 F0)
{
    // Ensure cosTheta is clamped to valid range
    cosTheta = ClampEpsilon(cosTheta);

    // Schlick's approximation
    return F0 + (1.0 - F0) * pow(1.0 - cosTheta, FresnelPower);
}

// Fresnel Reflectance for Conductors (Metals)
float3 FresnelConductor(float3 eta, float3 k, float cosTheta)
{
    // Ensure cosTheta is clamped to the valid range [0, 1]
    cosTheta = ClampEpsilon(cosTheta);
    float cosTheta2 = cosTheta * cosTheta;
    float sinTheta2 = ClampEpsilon(1.0 - cosTheta2);

    // Compute eta^2 and k^2
    float3 eta2 = ClampEpsilon(eta * eta);
    float3 k2 = (k * k);

    // Calculate the terms used in the Fresnel equations
    float3 temp = ClampEpsilon(eta2 - k2 - sinTheta2);
    float3 a2pb2 = (sqrt(temp * temp + 2.0 * eta2 * k2));

    // Calculate Rs and Rp
    float3 Rs = ((a2pb2 - temp) / (a2pb2 + temp + 2.0 * cosTheta2));
    float3 Rp = ((a2pb2 * cosTheta2 - sinTheta2) / (a2pb2 * cosTheta2 + sinTheta2));

    // Average Rs and Rp to get the Fresnel reflectance
    return ClampEpsilon(0.5 * (Rs + Rp));
}

// Complex Index of Refraction (Sellmeier-based for dielectrics)
float2 ComplexIOR_Dielectric(MaterialProperties mat, float wavelength)
{
    float lambda2 = wavelength * wavelength;
    float eta2 = 1.0; // Initialize squared index of refraction

    for (int i = 0; i < 3; i++)
    {
        eta2 += mat.sellmeierCoefficients[i].B * lambda2 / (lambda2 - mat.sellmeierCoefficients[i].C);
    }

    float eta = sqrt(eta2);
    float k = epsilon; // No absorption for pure dielectrics
    return (float2(eta, k));
}


// Complex Index of Refraction (Drude model for conductors)
float2 ComplexIOR_Conductor(MaterialProperties mat, float wavelength)
{
    float omegaP2 = mat.plasmaFrequency * mat.plasmaFrequency;
    float gamma2 = mat.collisionFrequency * mat.collisionFrequency;
    float lambda2 = wavelength * wavelength;

    float realPart = 1.0 - omegaP2 * lambda2 / (gamma2 + lambda2);
    float imaginaryPart = omegaP2 * wavelength / (mat.collisionFrequency * (mat.collisionFrequency + wavelength * wavelength));

    float eta = sqrt(0.5 * (realPart + sqrt(realPart * realPart + imaginaryPart * imaginaryPart)));
    float k = sqrt(0.5 * (-realPart + sqrt(realPart * realPart + imaginaryPart * imaginaryPart)));

    return float2(eta, k);
}

// Combined Fresnel Calculation (Handles Both Dielectrics and Conductors)
float3 FresnelSchlick(float3 F0, float cosTheta, float metalness, float3 k)
{
    // Interpolate between dielectric and conductor Fresnel based on metalness
    return ClampEpsilon(slerp(FresnelSchlickDielectric(cosTheta, F0),
               FresnelConductor(F0, k, cosTheta),
               metalness));
}

// Enhanced QFresnel Function (Uses Improved Fresnel Calculation)
float4 QFresnelm(float4 color, float metalness)
{
    float3 F0 = slerp(vec3(0.04, 0.04, 0.04), ClampEpsilon(color.rgb), ClampEpsilon(metalness));
    float3 fresnel = ClampEpsilon(FresnelSchlick(F0, ClampEpsilon(config.HdotV), ClampEpsilon(metalness), epsilon));

    // Interpolate between diffuse and specular Fresnel based on NdotV
    return float4(slerp(ClampEpsilon(color.rgb) * (1.0 - fresnel), fresnel, saturate(1.0 - ClampEpsilon(config.NdotV))), 1.0);
}


// Fresnel Reflectance for Dielectric Materials (Unpolarized)
float3 FresnelDielectric(float etaI, float etaT, float cosThetaI)
{
    // Ensure cosThetaI is clamped to valid range
    cosThetaI = ClampEpsilon11(cosThetaI);
    etaI = (etaI);
    etaT = (etaT);
    // Handle case where ray is inside the material
    if (cosThetaI < 0.0)
    {
        float f = etaI;
        etaI = etaT;
        etaT = f;

        cosThetaI = -cosThetaI;
    }

    // Calculate cosine of transmission angle using Snell's law
    float sinThetaTSqr = ClampEpsilon((etaI / etaT) * (etaI / etaT) * (1.0 - cosThetaI * cosThetaI));
    float cosThetaT = ClampEpsilon(sqrt(max(0.0, 1.0 - sinThetaTSqr))); // Handle possible total internal reflection

    // Fresnel equations (unpolarized)
    float rParallel = ((etaT * cosThetaI) - (etaI * cosThetaT)) / ((etaT * cosThetaI) + (etaI * cosThetaT));
    float rPerpendicular = ((etaI * cosThetaI) - (etaT * cosThetaT)) / ((etaI * cosThetaI) + (etaT * cosThetaT));

    // Average reflectance for unpolarized light
    return (0.5 * (rParallel * rParallel + rPerpendicular * rPerpendicular));
}

#define NUM_WAVELENGTH_SAMPLES 32  // Increase for higher accuracy

float4 QFresnel(float4 color, MaterialProperties mat)
{
    // Extract relevant material properties
    float3 F0 = slerp(vec3(0.04, 0.04, 0.04), max(vec3(0.04, 0.04, 0.04)
    , mat.baseColor), mat.metalness);
    float roughness = 1.0 - ClampEpsilon(mat.smoothness);

    // Wavelength range (visible spectrum)
    const float minWavelength = 380e-9; // Violet (in meters)
    const float maxWavelength = 750e-9; // Red (in meters)

    // Calculate wavelength step
    float wavelengthStep = (maxWavelength - minWavelength) / float(NUM_WAVELENGTH_SAMPLES - 1);

    // Accumulate Fresnel contributions across wavelengths
    float3 fresnelSum = float3(0.0, 0.0, 0.0);
    float3 specularSum = float3(0.0, 0.0, 0.0);
    for (int i = 0; i < NUM_WAVELENGTH_SAMPLES; ++i)
    {
        float wavelength = minWavelength + float(i) * wavelengthStep;

            // Metal: Use complex IOR-based Fresnel
        float2 etaK = ComplexIOR_Conductor(mat, wavelength);
        float fresnel1 = FresnelConductor(etaK.x, etaK.y, config.HdotV);
        
        float etaI = 1.0;
        float etaT = ComplexIOR_Dielectric(mat, wavelength);
        //float etaT = CalculateRefractiveIndexUsingSellmeier(mat, wavelength); // Transmitted medium refractive index
        float fresnel2 = FresnelDielectric(etaI, etaT, config.HdotV);
        
        float fresnel = slerp(ClampEpsilon(fresnel1), ClampEpsilon(fresnel2), ClampEpsilon(mat.metalness));

        // Adjust Fresnel for roughness (for dielectrics only)
        if (mat.metalness == 0.0)
        {
            float3 F90 = F0 * (1.0 + roughness * roughness);
            fresnel = ClampEpsilon(slerp(fresnel, F90, pow(1.0 - config.NdotV, FresnelPower)));
        }

        fresnelSum += fresnel * wavelengthStep; // Sum Fresnel contributions, weighted by wavelength interval
    
        // Calculate and accumulate specular reflection (assuming a Cook-Torrance microfacet model)
        float alpha = roughness * roughness;
        float D = ClampEpsilon(GGXDistribution(ClampEpsilon(config.NdotH), alpha));
        float G = ClampEpsilon(SmithGGXCorrelated(ClampEpsilon(config.NdotL), ClampEpsilon(config.NdotV), roughness * roughness));
        specularSum += (D * G * fresnel) * wavelengthStep; // Specular contribution
    }
    
    // Normalize the sums
    fresnelSum /= (maxWavelength - minWavelength);
    specularSum /= (maxWavelength - minWavelength);

    // Combine diffuse, specular, and Fresnel
    float3 diffuse = ClampEpsilon(color.rgb) * ClampEpsilon(1.0 - fresnelSum) * ClampEpsilon(1.0 - mat.metalness);
    return float4(ClampEpsilon(diffuse) + ClampEpsilon(specularSum), 1.0);
}

float SigmoidFalloff(float dist, float sigmoid_range, float sigmoid_sharpness)
{
    float x = dist / sigmoid_range;
    return 1.0 / (1.0 + exp(sigmoid_sharpness * (x - 1.0)));
}


// Main Lighting Calculation
float3 FresnelSchlick(float cosTheta, float3 F0)
{
    F0 = ClampEpsilon(F0);
    cosTheta = ClampEpsilon(cosTheta);
    return F0 + (1.0 - F0) * pow(1.0 - cosTheta, FresnelPower);
}

float3 CalculateHemisphereLighting(float3 normal, float3 upColor, float3 downColor)
{
    float3 hemisphereLighting = float3(0, 0, 0);
    
    // Upward hemisphere
    hemisphereLighting += upColor * max(0, normal.y);
    
    // Downward hemisphere
    hemisphereLighting += downColor * max(0, -normal.y);
    
    return hemisphereLighting;
}

float3 CalculateGlobalIllumination(float3 pixelPos, float3 normal, float3 upColor, float3 downColor, float3 ambientColor, float ambientIntensity)
{
    float3 gi = float3(0, 0, 0);
    
    // Ambient term
    gi += ambientColor * ambientIntensity;
    
    // Hemisphere lighting
    float3 hemisphereLighting = config.NdotV - CalculateHemisphereLighting(normal, upColor, downColor);
    gi += hemisphereLighting;
    
    return gi;
}

float3 CalculateLighting(
    float3 normal,
    float3 lightPos,
    float3 pixelPos,
    float shininess,
    float3 ambientColor,
    float3 diffuseColor,
    float3 specularColor,
    float3 lightColor,
    float lightIntensity,
    float screenGamma,
    MaterialProperties mat,
    float2 uv,
    float2 uvOffset)
{
    normal = normalize(normal);
    float3 lightDir = normalize(lightPos - pixelPos);
    float3 viewDir = normalize(config.viewPos - pixelPos);
    
    float distance = length(lightPos - pixelPos);
    float attenuation = 1.0 / (distance * distance) * (config.depth * 0.5 + 0.5);
    
    float3 halfVec = normalize(lightDir + viewDir);
    float NdotL = ClampEpsilon(dot(normal, lightDir));
    float NdotV = ClampEpsilon(dot(normal, viewDir));
    float NdotH = ClampEpsilon(dot(normal, halfVec));
    float VdotH = ClampEpsilon(dot(viewDir, halfVec));
    
    // Specular Distribution
    float roughness2 = (mat.smoothness * mat.smoothness);
    float D = GGXDistribution(NdotH, roughness2);
    float G = SmithGGXCorrelated(NdotL, NdotV, roughness2);
    
    // Fresnel
    float3 F0 = slerp(diffuseColor, mat.baseColor, mat.metalness);
    float3 F = ClampEpsilon(FresnelSchlick(NdotV, F0));
    
    // Combine into final color
    float3 diffuse = diffuseColor * (1 - F) * mat.baseColor / PI;
    float3 specular = saturate((max(0.1, D) * max(0.1, G) * max(0.1, F)) * lightColor * lightIntensity * attenuation);
    float3 color = saturate(diffuse + specular);
    
    // Apply gamma correction
    return pow(
        (slerp(
            mir2D(diffuseMap, config.uv),
            mir2D(rtMap1, config.uv), 0.5).xyz +
        specular) * (1 + CalculateGlobalIllumination(pixelPos, normal, color, float3(1, 1, 1), ambientColor, 1)), 1.0 / screenGamma);
}

float4 CalculateMaterialLighting(
    float3 diffuse,
    float3 pixelPos,
    float3 normal,
    float3 viewDir,
    MaterialProperties mat,
    float3 attenuation,
    float aberrationStrength,
    float2 uv,
    float2 uvOffset)
{
    normal = normalize(normal);
    float3 lightDir = normalize(config.sunPos - pixelPos);
    
    float distance = length(config.sunPos - pixelPos);
    distance *= distance; // Square distance for attenuation
    
    float3 halfVec = normalize(lightDir + viewDir);
    float NdotL = ClampEpsilon(dot(normal, lightDir));
    float NdotV = ClampEpsilon(dot(normal, viewDir));
    float NdotH = ClampEpsilon(dot(normal, halfVec));
    float VdotH = ClampEpsilon(dot(viewDir, halfVec));
    
    float4 lighting = float4(CalculateLighting(normal, config.sunPos, pixelPos, aberrationStrength, slerp(float3(0.02, 0.02, 0.02), float3(0.07, 0.07, 0.07), 1 - config.depth), diffuse, mat.baseColor, diffuse + mat.baseColor, 0.5, 1, mat, uv, uvOffset), 1);
    
    return float4(diffuse, 1) * lighting;
}

float3 RefractRay(float3 rayDirection, float3 normal, float refractiveIndex)
{
    // Ensure vectors are normalized
    rayDirection = normalize(rayDirection);
    normal = normalize(normal);

    float cosI = ClampEpsilon11(dot(-rayDirection, normal));

    // Handle edge cases for incident angles
    if (abs(cosI - 1.0) < epsilon || abs(cosI + 1.0) < epsilon)
        return reflect(rayDirection, normal); // Grazing incidence or facing directly

    float sinT2 = refractiveIndex * refractiveIndex * (1.0 - cosI * cosI);
    if (sinT2 > 1.0) // Total internal reflection
        return reflect(rayDirection, normal);

    float cosT = sqrt(max(0.0, 1.0 - sinT2));
    return refractiveIndex * rayDirection + (refractiveIndex * cosI - cosT) * normal;
}

float3 RefractRayByAngle(float3 incidentRay, float3 normal, float refractedAngle)
{
    float3 refractedRay;
    float cosIncidentAngle = ClampEpsilon11(dot(-incidentRay, normal));
    float sinRefractedAngle = sin(refractedAngle);

    if (cosIncidentAngle < 0)
    {
        // Handle total internal reflection if necessary
        cosIncidentAngle = -cosIncidentAngle;
        normal = -normal;
    }

    float sinIncidentAngle = sqrt(1.0 - cosIncidentAngle * cosIncidentAngle);
    if (sinIncidentAngle / sinRefractedAngle > 1.0)
    {
        // Total internal reflection
        refractedRay = reflect(incidentRay, normal);
    }
    else
    {
        float cosRefractedAngle = sqrt(1.0 - sinRefractedAngle * sinRefractedAngle);
        refractedRay = sinRefractedAngle * -normalize(cross(normal, cross(-normal, incidentRay))) - cosRefractedAngle * normal;
    }

    return refractedRay;
}

float CalculateIncidentAngle(float3 incidentRay, float3 normal)
{
    return ClampEpsilon11(dot(-normalize(incidentRay), normalize(normal)));
}

float CalculateRefractedAngle(float3 incidentRay, float3 normal, float refractiveIndex1, float refractiveIndex2)
{
    // Validate refractive indices
    if (refractiveIndex1 <= 0.0 || refractiveIndex2 <= 0.0)
        return 0.0; // Invalid indices return zero angle

    // Normalize inputs
    normal = normalize(normal);
    incidentRay = normalize(incidentRay);

    // Calculate cosine of the incident angle using the dot product
    float cosIncidentAngle = dot(-incidentRay, normal);

    // Edge cases for incident angle
    if (abs(cosIncidentAngle) < epsilon)
        return 0.0; // Grazing incidence
    if (abs(cosIncidentAngle - 1.0) < epsilon)
        return PI / 2; // Perpendicular incidence

    // Calculate sine of incident and refracted angles
    float sinIncidentAngle = sqrt(max(0.0, 1.0 - cosIncidentAngle * cosIncidentAngle));
    float sinRefractedAngle = (refractiveIndex1 / refractiveIndex2) * sinIncidentAngle;

    // Check for total internal reflection
    if (sinRefractedAngle > 1.0)
        return PI / 2; // Angle of 90 degrees for total internal reflection

    float cosRefractedAngle = sqrt(max(0.0, 1.0 - sinRefractedAngle * sinRefractedAngle));
    return acos(clamp(cosRefractedAngle, -1.0, 1.0));
}

// Linear interpolation function
float LinearInterpolate(float x, float x0, float y0, float x1, float y1)
{
    return y0 + (x - x0) * (y1 - y0) / (x1 - x0);
}

float3 RefractRay(float3 incidentRay, float3 normal, float n1, float n2)
{
    float cosI = dot(-normalize(incidentRay), normalize(normal));
    float sinT2 = (n1 / n2) * (n1 / n2) * (1.0 - cosI * cosI);
    if (sinT2 > 1.0)
        return reflect(incidentRay, normal); // Total internal reflection condition
    float cosT = sqrt(1.0 - sinT2);
    return normalize((n1 / n2) * incidentRay + (n1 / n2 * cosI - cosT) * normal);
}

float4 CalculateOcclusionFactor(vec2 oosz, vec2 uv, float d, float normalOcclusionFactor, float2 gradientModulation, GradientModulationConfig config)
{
    float sd = (d);
    float4 ret = 0;
    float tot = 0;
    float dist = 1;
    float diffRange = (sd - OCCLUSION_MIN_DIFFERENCE);
    [unroll(10)]
    for (int dx = -2; dx <= 2; ++dx)
    {
        [unroll(10)]
        for (int dy = -2; dy <= 2; ++dy)
        {
            if (dx == 0 || dy == 0)
                continue;
            
            float weight = 1 / (1 + length(float2(dx, dy)));
            float2 offset = float2(dx, dy) * (dist / 2) * weight;
            
            float td = dep2D(oosz, depthMap, uv + offset * oosz * gradientModulation);
            if (td < diffRange)
            {
                ret += weight * (sd - td);
                tot += weight;
            }
        }
    }
    return tot == 0 ? 0 : ClampEpsilon11(ret / tot);
}

// Gaussian bloom implementation
float4 GaussianBloom(float2 oosz, Texture2D tex, float2 uv, float dist, float threshold, float4 minColor, float3 power, float sigma)
{
    // Sample the original texture
    float4 color = mir2D(tex, uv);

    // Initialize variables for Gaussian blur
    float4 blurColor = float4(0.0, 0.0, 0.0, 0.0);
    float totalWeight = 0.0;
    int radius = min(10, max(1, dist / 2));
    
    // Apply Gaussian blur in a 3x3 kernel around the pixel
    [unroll(10)]
    for (int dx = -radius; dx <= radius; ++dx)
    {
        [unroll(10)]
        for (int dy = -radius; dy <= radius; ++dy)
        {
            if (dx == 0 || dy == 0)
                continue;
           
            float2 offset = float2(dx, dy) * dist;
            float weight = Gaussian(length(offset), sigma);
            blurColor += weight * mir2D(tex, uv + offset * oosz);
            totalWeight += weight;
        }
    }
    blurColor /= totalWeight;

    // Dynamic blending based on brightness and threshold
    float brightness = max(max(blurColor.r, blurColor.g), blurColor.b);
    if (brightness >= threshold)
    {
        float blendedColorR = slerp(color.x, max(minColor.x, (blurColor.x + brightness)), power.x);
        float blendedColorG = slerp(color.y, max(minColor.y, (blurColor.y + brightness)),
           power.y);
        float blendedColorB = slerp(color.z, max(minColor.z, (blurColor.z + brightness)), power.z);
        
        return float4(float3(blendedColorR, blendedColorG, blendedColorB), 1);
    }

    return color; // Return the original color if below the threshold
}



// bloom2 function
float4 bloom2(float2 oosz, Texture2D tex, float2 uv, float dist, float threshold, float2 modulation, float2 gradient)
{
  // Sample the original texture
    float4 color = mir2D(tex, uv);

  // Blur the color
    float4 blurColor = mir2D(tex, uv + float2(-dist, -dist) * 0.5f * oosz * modulation)
    + mir2D(tex, uv + float2(dist, dist) * 0.5f * oosz * modulation)
    + mir2D(tex, uv + float2(-dist, dist) * 0.5f * oosz * modulation)
    + mir2D(tex, uv + float2(dist, -dist) * 0.5f * oosz * modulation)
    + mir2D(tex, uv + float2(-dist, 0.0f) * oosz * modulation)
    + mir2D(tex, uv + float2(dist, 0.0f) * oosz * modulation)
    + mir2D(tex, uv + float2(0.0f, -dist) * oosz * modulation)
    + mir2D(tex, uv + float2(0.0f, dist) * oosz * modulation);

  // Average the blur samples
    blurColor *= 0.5 / 8.0;

  // Blend with original color if exceeds threshold
    if (max(max(color.r, color.g), color.b) > threshold)
    {
        color += blurColor;
    }

    return ClampEpsilon(color);
}

    vec2 leftUV = uv + ofsL;
    vec2 rightUV = uv + ofsR;
    
    // Sample the scene from the perspective of each eye
    float leftEyeColor = mir2D(tex, leftUV).r;
    float rightEyeColor = mir2D(tex, rightUV).b;
    
    vec4 ret = vec4(slerp(diffuseColor.xyz, vec3(leftEyeColor,
            diffuseColor.g,
            rightEyeColor), strength),
            1);
    
   
    return ret;
}

float RadicalInverse(uint x)
{
    // Reverse the bits of x
    x = (x << 16) | (x >> 16);
    x = ((x & 0x00ff00ff) << 8) | ((x & 0xff00ff00) >> 8);
    x = ((x & 0x0f0f0f0f) << 4) | ((x & 0xf0f0f0f0) >> 4);
    x = ((x & 0x33333333) << 2) | ((x & 0xcccccccc) >> 2);
    x = ((x & 0x55555555) << 1) | ((x & 0xaaaaaaaa) >> 1);
    // Divide by 2^32
    return float(x) * 2.3283064365386963e-10;
}

}

float SSAO(float2 oosz, Texture2D<float> depthTex, float2 uv, float radius, float AO_DEPTH_THRESHOLD)
{
    float depthCenter = dep2D(oosz, depthTex, uv);
    float ao = 0.0;

    for (int x = -radius; x <= radius; x++)
    {
        for (int y = -radius; y <= radius; y++)
        {
            float2 offset = float2(x, y) * oosz;
            float sampleDepth = dep2D(oosz, depthTex, uv + offset);
            ao += (sampleDepth >= depthCenter + AO_DEPTH_THRESHOLD) ? 1 : 0;
        }
    }
    return 1.0 - (ao / ((2 * radius + 1) * (2 * radius + 1)));
}

float3 EdgeDetectionGlow(Texture2D tex, float2 uv, float edgeIntensity)
{
    // Sobel filter kernel definitions
    float3x3 sobelX = float3x3(-1, 0, 1, -2, 0, 2, -1, 0, 1);
    float3x3 sobelY = float3x3(-1, -2, -1, 0, 0, 0, 1, 2, 1);

    float3 edgeColor = float3(0.0, 0.0, 0.0);
    float3 normalColor = mir2D(tex, uv).rgb;
    float edgeDetectionThreshold = 0.1; // Threshold for edge detection

    // Sobel edge detection
    float gx = 0.0;
    float gy = 0.0;
    [unroll(10)]
    for (int x = -1; x <= 1; x++)
    {
        [unroll(10)]
        for (int y = -1; y <= 1; y++)
        {
            float3 sampleColor = mir2D(tex, uv + float2(x, y) * 0.002).rgb; // 0.002 is a sample offset value
            gx += ClampEpsilon(dot(sampleColor, sobelX[x + 1][y + 1]));
            gy += ClampEpsilon(dot(sampleColor, sobelY[x + 1][y + 1]));
        }
    }

    float edgeMagnitude = sqrt(gx * gx + gy * gy);
    bool isEdge = edgeMagnitude > edgeDetectionThreshold;

    // Apply glowing effect on edges
    if (isEdge)
    {
        // Glowing effect - this could be a neon or spectral color shift
        float glowStrength = sin(edgeMagnitude * edgeIntensity) * 0.5 + 0.5;
        edgeColor = slerp(normalColor, float3(0.0, 0.8, 1.0), glowStrength); // Example: Glowing cyan color
    }

    return isEdge ? edgeColor : normalColor;
}

// Improved Parallax Occlusion function
ParallaxResult ParallaxOcclusion(
    float2 oosz,
    Texture2D<float> depthMap,
    float2 uv,
    float2 offset,
    float2 offset2,
    int numLayers
)
{
    numLayers = max(1, numLayers);
    
    // Initialize the result structure
    ParallaxResult result = CreateParallaxResult();
    result.uv = uv;

    // Calculate the initial depth value
    result.depth = dep2D(depthMap, uv);
    
    // Initialize variables for the loop
    float layerDepth = result.depth;
    float depthStep = 1.0f / numLayers;

    // Pre-calculate gradients to avoid repetitive calculations
    float2 modulation = GetModulation(depthMap, uv);
    
    [unroll(10)]
    for (int i = 0; i < numLayers; i++)
    {
        float2 ofs1 = offset * modulation;
        float2 ofs2 = offset2 * modulation;
        
        float2 newUV = result.uv + slerp(ofs2, ofs1, result.depth * ((i + 1) / numLayers));

        // Calculate the new depth value
        float newDepth = dep2D(depthMap, newUV);

        // Check if the new depth value is less than or equal to the current depth value
        if (newDepth <= result.depth - 0.001)
        {
            result.depth = newDepth;
        }
        result.uv = newUV;
        modulation = GetModulation(depthMap, result.uv);
        // Update layer depth for the next iteration
        layerDepth += depthStep;
    }

    // Return the result structure
    return result;
}



// Function 2: Chromatic Aberration Simulation
float3 ChromaticAberration(float2 oosz, Texture2D tex, float3 color, float2 uv, float2 aberrationOffset)
{
    // Simulates chromatic aberration by slightly offsetting the color channels
    float2 redOffset = uv + float2(aberrationOffset.x, 0);
    float2 greenOffset = uv;
    float2 blueOffset = uv - float2(aberrationOffset.y, 0);
    float red = mir2D(tex, redOffset).r;
    float green = mir2D(tex, greenOffset).g;
    float blue = mir2D(tex, blueOffset).b;
    return saturate(float4(color * float3(red, green, blue), 1)).xyz;
}

float calculateFresnel(float surfaceDot, float indexOfRefraction)
{
    float cosi = ClampEpsilon(surfaceDot);
    float etai = 1.0;
    float etat = ClampEpsilon(indexOfRefraction);

    if (cosi > 0.0)
    {
        float temp = etai;
        etai = etat;
        etat = temp;
    }

    float sint = etai / etat * sqrt(max(epsilon, 1.0f - cosi * cosi));

    if (sint >= 1.0)
    {
        return 1.0;
    }

    float cost = sqrt(max(epsilon, 1.0f - sint * sint));
    cosi = abs(cosi);

    float Rs = ((etat * cosi) - (etai * cost)) / ((etat * cosi) + (etai * cost));
    float Rp = ((etai * cosi) - (etat * cost)) / ((etai * cosi) + (etat * cost));

    return float(Rs * Rs + Rp * Rp) / 2.0;
}

float HenyeyGreensteinPhaseFunction(float g, float cosTheta)
{
    float denom = 1.0 + g * g - 2.0 * g * cosTheta;
    return (1.0 / (4.0 * PI)) * ((1.0 - g * g) / (denom * sqrt(denom)));
}

float3 CalculateScattering(float3 position, float3 direction, float wavelength, Material material)
{
    float scatteringCoefficient = material.scatteringCoefficient;
    float cosTheta = ClampEpsilon(dot(direction, material.normal));
    float phaseFunction = HenyeyGreensteinPhaseFunction(material.g, cosTheta);
    return scatteringCoefficient * phaseFunction * direction;
}

float2 Hammersley2D(uint index, uint numSamples)
{
    float phi = frac(float(index) * 0.618033988749895);
    return float2(float(index) / float(numSamples), phi);
}

float StratifiedSampling(float3 position, float3 direction, float wavelength, Material material)
{
    float2 sample = Hammersley2D(stratifiedSampleIndex, NUM_STRATA);
    float phi = 2.0 * PI * sample.x;
    float cosTheta = 1.0 - 2.0 * sample.y;
    float sinTheta = sqrt(1.0 - cosTheta * cosTheta);
    float3 tangent = normalize(cross(material.normal, float3(1.0, 0.0, 0.0)));
    float3 bitangent = normalize(cross(material.normal, tangent));
    direction = normalize(cos(phi) * tangent + sin(phi) * bitangent + cosTheta * material.normal);
    float cosThetaI = ClampEpsilon(dot(direction, material.normal));
    return abs(cosThetaI) / PI;
}

float ImportanceSampling(float3 position, float3 direction, float wavelength, Material material)
{
    float3 normal = material.normal;
    float cosThetaI = ClampEpsilon(dot(direction, normal));
    return abs(cosThetaI) / PI;
}

bool IntersectRayPrimitive(float3 rayOrigin, float3 rayDirection, in Primitive primitive, inout float t)
{
    float3 oc = rayOrigin - primitive.position;
    float a = dot(rayDirection, rayDirection);
    float b = 2.0f * dot(oc, rayDirection);
    float c = dot(oc, oc) - primitive.radius * primitive.radius;
    float discriminant = b * b - 4.0f * a * c;

    if (discriminant < 0.0f)
    {
        return false;
    }

    float sqrt_discriminant = sqrt(discriminant);
    float t0 = (-b - sqrt_discriminant) / (2.0f * a);
    float t1 = (-b + sqrt_discriminant) / (2.0f * a);

    if (t0 > t1)
    {
        float temp = t0;
        t0 = t1;
        t1 = temp;
    }

    if (t0 < 0.0f)
    {
        if (t1 < 0.0f)
        {
            return false;
        }
        else
        {
            t = t1;
            return true;
        }
    }
    else
    {
        t = t0;
        return true;
    }
}

float Interpolate(const float values[81], int index, float offset)
{
    if (index < 0)
        return 0;
    if (index > 80)
        return 1;
    if (offset == 0 || index == 80)
        return values[index];

    float x0 = index * LEN_STEP;
    float x1 = x0 + LEN_STEP;
    float y0 = values[index];
    float y1 = values[index + 1];

    return slerp(y0, y1, offset / LEN_STEP);
}

float GammaCorrect_sRGB(float c)
{
    if (c <= 0.0031308)
        return 12.92 * c;
    float a = 0.055;
    return (1 + a) * pow(c, 1 / 2.2) - a;
}

float3 ConvertToRadiance(float len)
{
    if (len < LEN_MIN)
        return float3(0, 0, 0);
    if (len > LEN_MAX)
        return float3(1, 1, 1);
    len -= LEN_MIN;
    int index = int(floor(len / LEN_STEP));
    float offset = len - LEN_STEP * index;

    float x = Interpolate(X, index, offset);
    float y = Interpolate(Y, index, offset);
    float z = Interpolate(Z, index, offset);
    float3 rgb = mul(float3(x, y, z), MATRIX_SRGB_D65);
    rgb.r = saturate(GammaCorrect_sRGB(rgb.r));
    rgb.g = saturate(GammaCorrect_sRGB(rgb.g));
    rgb.b = saturate(GammaCorrect_sRGB(rgb.b));
    return rgb;
}

float ConvertToWavelength(float3 rgb)
{
    float3 hsv = RGBtoHSV(rgb);
    float hue = hsv.x;
    float wavelength;

    if (hue >= 0 && hue < 60)
    {
        wavelength = slerp(620.0, 590.0, hue / 60.0);
    }
    else if (hue >= 60 && hue < 180)
    {
        wavelength = slerp(590.0, 495.0, (hue - 60) / 120.0);
    }
    else if (hue >= 180 && hue < 240)
    {
        wavelength = slerp(495.0, 450.0, (hue - 180) / 60.0);
    }
    else if (hue >= 240 && hue < 300)
    {
        wavelength = slerp(450.0, 380.0, (hue - 240) / 60.0);
    }
    else
    {
        wavelength = slerp(380.0, 620.0, (hue - 300) / 60.0);
    }

    return wavelength;
}

float3 CalculateRadianceContribution(float3 position, float3 direction, float wavelength, Material material)
{
    float3 radianceContribution = 0.0;
    float reflectivity = ClampEpsilon(material.reflection);
    float transparency = ClampEpsilon(material.transmission);
    float angleFactor = ClampEpsilon(dot(direction, config.normal));
    radianceContribution = ClampEpsilon((reflectivity + transparency) * angleFactor);
    float wavelengthFactor = 1000.0;
    radianceContribution *= wavelengthFactor;
    return radianceContribution;
}

float3 CalculateExitantRadiance(float3 position, float3 direction, float wavelength, Material material)
{
    float3 exitantRadiance = material.emission * ConvertToRadiance(wavelength);
    float3 incidentRadiance = CalculateRadianceContribution(position, direction, wavelength, material);
    exitantRadiance += incidentRadiance * material.reflection + material.transmission;
    return exitantRadiance;
}

float3 CalculateInterference(float3 position, float3 direction, float wavelength, Material material)
{
    float3 pathDifference = position - material.referencePoint;
    float phaseDifference = 2.0 * PI * ClampEpsilon(dot(pathDifference, direction)) / max(0.0001, wavelength);
    float3 interference = cos(phaseDifference) * material.interferenceStrength;
    return interference;
}

float3 ReflectDirection(float3 direction, float3 normal)
{
    return direction - 2.0 * dot(direction, normal) * normal;
}

float3 CalculateQuantumPropagation(float3 position, float3 direction, float wavelength, float time)
{
    direction = normalize(direction);
    float wavevector = 2.0 * PI / max(0.0001, wavelength);
    float angularFrequency = SPEED_OF_LIGHT * wavevector;
    float3 newPosition = position + direction * SPEED_OF_LIGHT * time;
    float phase = dot(newPosition, wavevector) - angularFrequency * time;
    float3 waveFunction = cos(phase) * dot(direction, wavevector);
    return newPosition + waveFunction * max(0.0001, wavelength);
}

// Function to get a basis matrix from a given vector
void GetBasisFromVector(float3 v, out float3 vx, out float3 vy)
{
    float3 vz = abs(v.x) > 0.99 ? float3(0, 1, 0) : float3(1, 0, 0);
    vz = normalize(cross(vz, v));
    
    vx = normalize(cross(vz, v));
    vy = cross(v, vx);
}

float3x3 GetBasisFromVector(float3 v)
{
    float3 vx, vy;
    GetBasisFromVector(v, vx, vy);
    return float3x3(vx, vy, v);
}

// Function to generate a random ray direction in a hemisphere
float3 RandomRayInHemisphere(float3 normal)
{
    float2 randVal = randomNoise2(normal.xy + normal.z);
    float3 u = normalize(float3(randVal.x * 2 - 1, randVal.y * 2 - 1, 1));
    u = mul(u, GetBasisFromVector(normal));
    
    float len = sqrt(1 - u.z * u.z);
    return float3(u.xy * len, u.z);
}

        float rayLength = 0;
        
        // March along ray and accumulate photon hits
        for (int j = 0; j < 100; j++)
        {
            float3 samplePos = worldPos + rayDir * rayLength;
            float3 photon = mir2D(gratingMap, samplePos.xy).xyz;
            if (any(photon))
            {
                float dist = distance(samplePos, worldPos);
                if (dist < photonRadius * photonScale)
                {
                    radiance += photon / (dist * dist);
                }
            }
            rayLength += photonRadius * photonScale * 0.5;
        }
    }
    radiance /= photonRayCount;
    
    // Convert from RGB to spectral
    float3 spectral = 0;
    for (i = 0; i < 9; i++)
    {
        spectral += radiance * RGB2SpectralWeight[i];
    }
    return spectral;
}
float3 SpectralRadianceEstimate(Texture2D gratingMap, float photonRadius, float photonScale, float3 worldPos, float3 normal, int PhotonRayCount)
{
    float3 radiance = 0;
    for (int i = 0; i < PhotonRayCount; i++)
    {
        float3 rayDir = normalize(RandomRayInHemisphere(normal));
        float rayLength = 0;
        
        // March along ray and accumulate photon hits
        for (int j = 0; j < 100; j++)
        {
            float3 samplePos = worldPos + rayDir * rayLength;
            float3 photon = mir2D(gratingMap, samplePos.xy).xyz;
            if (photon.r > 0 || photon.g > 0 || photon.b > 0)
            {
                float dist = distance(samplePos, worldPos);
                if (dist < photonRadius * photonScale)
                {
                    radiance += photon / (dist * dist);
                }
            }
            rayLength += photonRadius * 0.5;
        }
    }
    radiance /= PhotonRayCount;
    
    // Convert from RGB to spectral
    float3 spectral = 0;
    for (i = 0; i < 9; i++)
    {
        spectral += radiance * RGB2SpectralWeight[i];
    }
    return spectral;
}

float3 getNoisePattern(float2 uv, float scale)
{
    float3 noise1v = noise3(uv * scale).rgb * 2 - 1;
    float3 noise2v = noise3(uv * scale * 0.5).rgb * 2 - 1;
    float3 noise3v = noise3(uv * scale * 0.25).rgb * 2 - 1;
    return (noise1v + noise2v + noise3v) * 0.333 * cosTime01(timr * 0.1) * sinTime01(timr * 0.05);
}

// Function to generate a flicker pattern
float getFlickerPattern(float2 uv, float time, float hologramFlickerSpeed, float hologramFlickerRandomness, float hologramFlickerIntensity)
{
    float flicker = sin(time * hologramFlickerSpeed + randomNoise2(uv * hologramFlickerRandomness).x * 6.28);
    flicker += sin(time * hologramFlickerSpeed * 0.5 + randomNoise2(uv * hologramFlickerRandomness * 0.5).y * 6.28) * 0.5;
    return saturate(flicker * hologramFlickerIntensity * sinTime01(timr * 0.05) * cosTime01(timr * 0.1));
}

// Function to generate a scanline pattern
float getScanlinePattern(float2 uv, float time, float hologramScanlineWidth, float hologramScanlineSpeed, float hologramScanlineIntensity)
{
    float scanline = abs(frac(uv.y * hologramScanlineWidth + time * hologramScanlineSpeed) * 2.0 - 1.0);
    scanline += abs(frac(uv.y * hologramScanlineWidth * 0.5 + time * hologramScanlineSpeed * 0.5) * 2.0 - 1.0) * 0.5;
    return saturate(scanline * hologramScanlineIntensity * cosTime01(TotalTime * 0.1) * sinTime01(timr * 0.05));
}

// Function to generate accommodation blur
float getAccommodationBlur(float depth, float hologramAccommodationBlur, float hologramAccommodationIntensity)
{
    float blurFactor = saturate(depth * hologramAccommodationBlur);
    blurFactor += saturate(depth * hologramAccommodationBlur * 0.5) * 0.5;
    return slerp(1.0, 0.0, blurFactor * hologramAccommodationIntensity * cosTime01(timr * 0.1) * sinTime01(timr * 0.05));
}

// Wigner Distribution Function (WDF)
float2 wignerDistributionFunction(float2 uv, float2 direction, float width)
{
    float2 wdf = float2(0.0, 0.0);
    float2 offset = uv - 0.5;
    float radius = length(offset);
    float angle = atan2(offset.y, offset.x);
    float gaussian1 = exp(-radius * radius / (width * width));
    float gaussian2 = exp(-radius * radius / (width * width));
    wdf.x = gaussian1 * cos(angle * direction.x) + sin(angle * direction.x) * 0.1;
    wdf.y = gaussian2 * sin(angle * direction.y) + cos(angle * direction.y) * 0.1;
    return wdf * cosTime01(timr * 0.1);
}

// Gaussian Double-Slit Experiment
float2 gaussianDoubleSlit(float2 uv, float2 slitPositions, float slitWidth, float2 diffractionDirection)
{
    float2 wdf1 = wignerDistributionFunction(uv, diffractionDirection, slitWidth);
    float2 wdf2 = wignerDistributionFunction(uv - slitPositions, diffractionDirection, slitWidth);
    float2 interference = float2(abs(wdf1.x * wdf2.x), abs(wdf1.y * wdf2.y));
    
    float quantumFluctuation = randomNoise2(uv * 0.01).x * 0.1;
    return interference * sinTime01(TotalTime * 0.05) * (1 + quantumFluctuation);
}
float4 ps_albedo(
    Texture2D<float4> diffuseMap, float2 uv,
    Texture2D<float4> diffuseMap2, float2 uv2,
    Texture2D<float4> diffuseMap3, float2 uv3,
    Texture2D<float3> normalMap)
{
    // Calculate the physically based lighting
    float3 normal = normal2D(normalMap, uv).xyz;
    float ndotl = saturate(dot(normal, config.pixelToSunDir));
    float ndoth = config.NdotH;
    float roughness = 0.25; // Roughness value
    float metallic = 0.9; // Metallic value
    
    // Interpolate diffuse colors from three texture maps
    float3 diffuse1 = mir2D(diffuseMap, uv).xyz;
    float3 diffuse2 = mir2D(diffuseMap2, uv2).xyz;
    float3 diffuse3 = mir2D(diffuseMap3, uv3).xyz;
    float3 diffuse = lerp(diffuse1, lerp(diffuse2, diffuse3, 0.05), 0.05) + (1 - metallic) * float3(0.9, 0.9, 0.8) * ndotl;
    
    // Calculate specular reflection
    float3 specular = metallic * (ndotl * 0.04 + (1 - roughness) * pow(ndoth, 5));
    
    // Combine diffuse and specular components
    float3 color = diffuse + specular;
    
    // Calculate the depth value
    float depth = config.depth;

    // Calculate the focus point and aperture
    float focusPoint = 1 * cosTime01(timr);
    float aperture = 2 + cosTime01(timr);

    // Calculate the blur amount
    float blurAmount = 0.0;
    if (depth < focusPoint - aperture || depth > focusPoint + aperture)
    {
        blurAmount = smoothstep(0.0, 1.0, clamp(abs(depth - focusPoint) / aperture, 0, 1));
    }

    // Apply depth of field blur
    color *= (1 - blurAmount);

    // Calculate the holographic effect
    float2 distort = sin((uv + TotalTime * timr) * 0.05) * 0.005;
    uv += distort;
    float4 hologram = mir2D(diffuseMap, uv);

    // Recalculate the blur amount based on depth
    if (depth < focusPoint - aperture || depth > focusPoint + aperture)
    {
        blurAmount = smoothstep(0.0, 1.0, clamp(abs(depth - focusPoint) / aperture, 0, 1));
    }

    // Combine the hologram and blur
    float4 color4 = float4(color, 1) * hologram * (1 - blurAmount);
    return slerp(float4(color, 1), color4, config.NdotV);
}



float4 PS_Holographic(
    float2 oosz,
    Texture2D<float4> diffuseMap, float2 uv,
    Texture2D<float4> diffuseMap2, float2 uv2,
    Texture2D<float> depthMap)
{
    // Calculate the holographic effect
    float2 distort = sin((uv + TotalTime * timr) * 0.05) * 0.05;
    uv += distort;
    float4 hologram = mir2D(diffuseMap, uv);

    // Calculate the depth and blur
    float depth = dep2D(depthMap, uv);
    float focusPoint = 10.0;
    float aperture = 5.0;
    float blurAmount = 0.0;
    if (depth < focusPoint - aperture || depth > focusPoint + aperture)
    {
        blurAmount = smoothstep(0.0, 1.0, clamp(abs(depth - focusPoint) / aperture, 0, 1));
    }

    // Additional distortion for the second hologram
    uv2 += (sin((uv2 + TotalTime * timr) * 0.2) + cos((uv2 + TotalTime * timr) * 0.2)) * 0.02;
    float4 hologram2 = mir2D(diffuseMap2, uv2);

    // Combine the holograms and blur
    float4 color = slerp(hologram * (1 - blurAmount), hologram2 * blurAmount, depth * 0.5 + (1 - config.HdotV) * 0.5);
    return color;
}

float4 PS_HolographicFresnelSchlick(Texture2D<float4> diffuseMap, float2 uv, Texture2D<float4> diffuseMap2, float2 uv2, float NdotV, float HdotV, float VdotL, float depth, float invDepth)
{
    // Holographic effect using wave theory
    float2 distort = sin((uv - 0.5 + TotalTime * timr)) * 0.005 * invDepth + cos((uv - 0.5 + TotalTime * timr)) * 0.025 * depth;
    uv += distort;
    float4 hologram = mir2D(diffuseMap, uv);

    // Fresnel-Schlick term using electromagnetic theory
    float dotProduct = NdotV;
    float fresnel = pow(1 - dotProduct, 5);

    // Schlick approximation using quantum mechanics
    float schlick = fresnel * (1 - hologram.a) + hologram.a;

    uv2 += sin((uv2 + TotalTime * timr) * 0.01) * 0.01 + cos((uv2 + TotalTime * timr) * 0.01) * 0.01 * depth;
    float4 hologram2 = mir2D(diffuseMap2, uv2);
    schlick += fresnel * (1 - hologram2.a) + hologram2.a;

    // Subtle glow effect using thermal radiation
    float glow = smoothstep(0.0, 1.0, dotProduct) + depth;

    // Combine the hologram and Schlick approximation using wave-particle duality
    float4 color = glow * float4(0.0, 0.0, 0.0, 0.0) + hologram * schlick;

    // Depth-based blur effect using fluid dynamics
    float focusPoint = 0;
    float aperture = 0.025;
    float blurAmount = 0.0;
    if (depth < focusPoint - aperture || depth > focusPoint + aperture)
    {
        blurAmount = cos(TotalTime) * 20 * HdotV * NdotV * smoothstep(0.0, 1.0, abs(depth - focusPoint) / aperture);
    }
    color.rgb += blurAmount * float3(0.05, 0.0, 0.05);

    // Modulation effect
    float2 depthGradient = float2(depth, invDepth);
    float2 modulation = GetModulation(depthMap, uv);

    return slerp(slerp(color, color * schlick, HdotV), glow * float4(0, 0, 0, 0), 1 - VdotL) * float4(modulation, 1.0, 1.0);
}

float4 QuantumFluctuation(float2 oosz, float2 uv, float3 viewDir, float3 lightDir, float3 normal, float3 diffuse)
{
	// Grating map
    float grating = dep2D(oosz, gratingDepth4, uv);
    float3 gratingNormal = normal2D(gratingNormal4, uv);
    float2 distortedUV = uv + (grating * 2.0 * 3.14159) / 10000;

	// Depth map
    float depth = dep2D(oosz, depthMap, uv);
    float3 position = float3(uv * 2.0 - 1.0, depth);

	// Normal map
    normal = normalize(normal);

	// Quantum fluctuation
    float quantum = ClampEpsilon(dot(gratingNormal, normal));
    float2 quantumUV = uv + (viewDir.xy / viewDir.z + normal.xy / normal.z + gratingNormal.xy / gratingNormal.z) * .5 * quantum;
    float3 quantumColor = lerp(mir2D(diffuseMap, uv).xyz, mir2D(gratingMap1, uv + .01 * refract(-config.viewDir.xy / config.viewDir.z, normal.xy / normal.z, length(uv - .5) * cosTime01(timr))).xyz, .1);

	// Calculate lighting
    float3 ambient = float3(0.1, 0.1, 0.1);
    float3 diffuseLight = ClampEpsilon(dot(normal, lightDir)) * diffuse;
    float3 glow = pow(ClampEpsilon(dot(reflect(-lightDir, normal), viewDir)), 10.0) * diffuse;

	// Combine effects
    float3 finalColor = (ambient + diffuseLight + glow) * diffuse;
    finalColor += quantumColor;

    return float4(saturate(finalColor), 1.0);
}


inline float2 zoom2D(float2 uv, float zoom)
{
    return (uv - .5) * zoom + .5;
}

inline float2 uvZoom(float2 uv, float scale)
{
    return zoom2D(uv, 1 - scale * dep2D(uv) * cosTime01(timr));
}


// Helper functions for time-based modulation factors
inline float cosTimeMod(float time, float scale, float frequency)
{
    return cos(time * frequency) * scale;
}

inline float sinTimeMod(float time, float scale, float frequency)
{
    return sin(time * frequency) * scale;
}

// Function to calculate rotation-aware offset
inline float2 CalculateRotationOffset(float2 uv, float2 center, float angle)
{
    float s = sin(angle);
    float c = cos(angle);

    // Translate point back to origin
    uv -= center;

    // Rotate point
    float xnew = uv.x * c - uv.y * s;
    float ynew = uv.x * s + uv.y * c;

    // Translate point back
    uv = float2(xnew, ynew) + center;

    return uv;
}
